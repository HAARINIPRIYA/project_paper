───────────────────────────────────────────────────────────────
import requests
import json

BASE = "http://localhost:8000"

# 1. Health check
r = requests.get(f"{BASE}/health")
print("Health:", r.json())

# 2. List models
r = requests.get(f"{BASE}/models")
print("Models:", r.json()["count"], "available")

# 3. Simple prediction
payload = {
    "Planting_Date": "2023-06-15",
    "Variety": "Co-0238",
    "Soil_Type": "Loamy"
}
r = requests.post(f"{BASE}/predict", json=payload)
print("Auto-predict:", r.json())

# 4. CatBoost prediction
r = requests.post(f"{BASE}/predict/catboost", json={
    "Planting_Date": "2023-06-15",
    "Harvesting_Date": "2024-01-10",
    "Variety": "Co-0238",
    "Soil_Type": "Loamy",
    "Irrigation_Type": "Drip",
    "Fertilizer_Type": "Urea"
})
print("CatBoost:", r.json())

# 5. Ensemble prediction
r = requests.post(f"{BASE}/predict/ensemble", json={
    "records": [{
        "Planting_Date": "2023-06-15",
        "Variety": "Co-0238",
        "Soil_Type": "Loamy"
    }]
})
print("Ensemble:", r.json())
